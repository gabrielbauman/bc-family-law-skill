\newcommand{\authority}[3]{%
    \newpage
    \phantomsection
    \label{auth:#1}
    \addcontentsline{toc}{section}{Tab #1: #2}
    \begin{center}
        \vspace*{3cm}
        {\Huge\bfseries Tab #1}

        \vspace{2em}
        {\Large\itshape #2}
    \end{center}
    \includepdf[pages=-]{#3}
}
