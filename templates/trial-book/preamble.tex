% ============================================
% NUMBERED PARAGRAPHS (court-style)
% Use \para at the start of each paragraph of testimony or argument so
% it can be referenced ("paragraph 14") in court. \resetpara restarts
% numbering at each major section.
% ============================================
\newcounter{para}
\newcommand{\para}{\refstepcounter{para}\textbf{\thepara.}\quad}
\newcommand{\resetpara}{\setcounter{para}{0}}

% ============================================
% EVIDENCE QUOTES
% Block-quote environment for quoted evidence; keeps quotes visually
% distinct from narrative.
% ============================================
\newenvironment{evidencequote}{\begin{quote}\itshape}{\end{quote}}

% ============================================
% TAB MACHINERY (evidence binder)
% \tabref{A-1}            -> clickable "Tab A-1" reference in the text
% \tabsection{A}{Title}{} -> cover page for a tab group, with TOC entry
% \exhibitheader{A-1}{Title}{Source}{Date}
%                         -> standard exhibit header page. Source and
%                            date ONLY — no summaries or commentary:
%                            descriptions of what an exhibit shows are
%                            argument, and argument does not belong in
%                            the evidence binder.
% ============================================
\newcommand{\tabref}[1]{\hyperref[tab:#1]{\textbf{Tab~#1}}}

\newcommand{\tabsection}[3]{%
    \newpage
    \phantomsection
    \addcontentsline{toc}{section}{Tab #1: #2}
    \begin{center}
        {\Huge\bfseries Tab #1}

        \vspace{1em}
        {\LARGE #2}
    \end{center}
    \vspace{2em}
    #3
}

\newcommand{\exhibitheader}[4]{%
    \newpage
    \phantomsection
    \label{tab:#1}
    \addcontentsline{toc}{subsection}{Tab #1: #2}
    \begin{center}
        {\LARGE\bfseries Tab #1}

        \vspace{1em}
        {\Large #2}

        \vspace{1em}
        \begin{tabular}{ll}
            \textbf{Source:} & #3 \\
            \textbf{Date:} & #4 \\
        \end{tabular}
    \end{center}
    \vspace{1em}
}
